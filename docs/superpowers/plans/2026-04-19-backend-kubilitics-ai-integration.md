# Backend ↔ kubilitics-ai Integration (Subproject 3a) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Wire `kubilitics-backend` to talk to the existing `kubilitics-ai` brain via gRPC streaming + HTTP control. Define `agent_runtime.proto` (the new typed contract for backend→AI direction) and implement a thin LLM-direct stub of `AgentRuntimeService` so the full vertical (UI → backend → kubilitics-ai → LLM → tokens back) works end-to-end. Retire the parallel re-implementations (supervisor, kotg-schema, kotg-ai-server, NoOpGate).

**Architecture:** Two repos. `vellankikoti/kotg.ai/kubilitics-ai/` defines `api/proto/v1/agent_runtime.proto` and implements `AgentRuntimeService` (LLM-direct path only in this slice; Router/Python/kagent come in 3b-3d). `vellankikoti/kubilitics/kubilitics-backend/` adds `internal/ai/aiclient/` (gRPC + HTTP clients), rewires `internal/ai/handlers/chat.go` to drive WS frames from `AIEvent` streams, archives the supervisor stack, replaces `cfg.AI.binary_path/idle_shutdown_seconds/...` with `cfg.AI.endpoint/http_endpoint`. Chat panel UI is unchanged.

**Tech Stack:** Go 1.24+, `google.golang.org/grpc`, `google.golang.org/protobuf`, buf for proto generation. Existing kubilitics-ai LLM provider clients reused (Anthropic/Ollama/custom). Test harnesses: `bufconn` (in-process gRPC for backend tests), in-process gRPC server (for kubilitics-ai stub used by manual smoke).

**Spec:** `docs/superpowers/specs/2026-04-19-backend-kubilitics-ai-integration-design.md`. Read it first.

**Push policy:** push only to `vellankikoti/kubilitics` and `vellankikoti/kotg.ai`. Never to any kubilitics/* org repo.

package kubilitics.ai.v1;

option go_package = "github.com/kubilitics/kubilitics-ai/api/proto/v1;aiv1";

import "google/protobuf/timestamp.proto";
import "google/protobuf/empty.proto";

// AgentRuntimeService is the single backend→AI contract. Streaming RPCs
// emit AIEvent frames (the Output Normalizer contract).
service AgentRuntimeService {
  rpc Chat(stream ChatRequest) returns (stream AIEvent);
  rpc RunAgent(stream AgentRequest) returns (stream AIEvent);
  rpc CancelTurn(CancelRequest) returns (google.protobuf.Empty);
  rpc Capabilities(google.protobuf.Empty) returns (CapabilitiesResponse);
}

// ── ChatRequest (client → server) ─────────────────────────────────────
message ChatRequest {
  oneof request {
    CreateSession create = 1;
    UserMessage   message = 2;
  }
}

message CreateSession {
  string focus_cluster_id = 1;
  string user_id = 2;
  string title = 3;
}

message UserMessage {
  string session_id = 1;
  string turn_id = 2;
  string text = 3;
  string context_hint = 4;
}

// ── AgentRequest (client → server, for RunAgent) ──────────────────────
message AgentRequest {
  oneof request {
    AgentInvoke invoke = 1;
    AgentInput  input = 2;
  }
}

message AgentInvoke {
  string agent_id = 1;
  string focus_cluster_id = 2;
  string user_id = 3;
  string trace_id = 4;
  map<string, string> parameters = 5;
}

message AgentInput {
  string run_id = 1;
  string text = 2;
}

message CancelRequest {
  oneof target {
    string session_id = 1;
    string run_id = 2;
  }
}

// ── AIEvent — THE OUTPUT NORMALIZER CONTRACT ──────────────────────────
message AIEvent {
  string anchor_id = 1;
  google.protobuf.Timestamp emitted_at = 2;
  // Bumped on any oneof addition/removal/renumbering. Backend MUST refuse
  // events whose schema_version is incompatible with what it was built for.
  string schema_version = 99;

  oneof event {
    TextDelta      text_delta      = 10;
    ToolStart      tool_start      = 11;
    ToolEnd        tool_end        = 12;
    ActionPending  action_pending  = 13;
    PlanProposed   plan_proposed   = 14;
    Citation       citation        = 15;
    Done           done            = 16;
    ErrorEvent     error_event     = 17;
    SessionCreated session_created = 18;
  }
}

message TextDelta {
  string text = 1;
}

message ToolStart {
  string tool_call_id = 1;
  string tool_name = 2;
  string preview = 3;
}

message ToolEnd {
  string tool_call_id = 1;
  bool ok = 2;
  string preview = 3;
}

message ActionPending {
  string proposal_id = 1;
  ActionTier tier = 2;
  string action_type = 3;
  string target = 4;
  string dry_run_preview = 5;
  bytes payload_json = 6;
  google.protobuf.Timestamp expires_at = 7;
}

enum ActionTier {
  ACTION_TIER_UNSPECIFIED = 0;
  ACTION_TIER_LOW = 1;
  ACTION_TIER_MEDIUM = 2;
  ACTION_TIER_HIGH = 3;
}

message PlanProposed {
  string plan_id = 1;
  string summary = 2;
  int32 step_count = 3;
}

message Citation {
  string assistant_text_anchor_id = 1;
  string tool_call_id = 2;
  string short_label = 3;
}

message Done {
  bool cancelled = 1;
  bool partial = 2;
  int32 prompt_tokens = 3;
  int32 completion_tokens = 4;
  // engine_used is INTERNAL ONLY. Backend's WS adapter MUST strip before

  string engine_used = 5;

}

  string code = 1;
  string message = 2;
}

  string session_id = 1;
}

message CapabilitiesResponse {
  string schema_version = 1;
  string ai_version = 2;
  repeated string engines = 3;
  repeated string providers = 4;
  repeated string models = 5;

// fakeLLM implements the minimum LLM-provider surface the runtime needs.
type fakeLLM struct{ chunks []string }

func (f *fakeLLM) StreamCompletion(ctx context.Context, _ string) (<-chan string, error) {
    out := make(chan string, len(f.chunks))
    go func() {
        defer close(out)
        for _, c := range f.chunks {
            select {
            case out <- c:
            case <-ctx.Done():
                return
            }
        }
    }()
    return out, nil
}

func newTestServer(t *testing.T, llm LLMProvider) (aiv1.AgentRuntimeServiceClient, func()) {
    t.Helper()
    lis := bufconn.Listen(1 << 20)
    s := grpc.NewServer()
    aiv1.RegisterAgentRuntimeServiceServer(s, New(Config{LLM: llm, AIVersion: "test", SchemaVersion: "1.0.0"}))
    go s.Serve(lis)

    conn, err := grpc.NewClient("passthrough:///bufnet",
        grpc.WithContextDialer(func(_ context.Context, _ string) (net.Conn, error) { return lis.Dial() }),

        t.Fatalf("dial: %v", err)

func TestChatStreamsTextDeltaThenDone(t *testing.T) {
    cli, cleanup := newTestServer(t, &fakeLLM{chunks: []string{"hello ", "world"}})
    defer cleanup()

    ctx := context.Background()
    stream, err := cli.Chat(ctx)
    if err != nil {

    if err != nil {
        t.Fatalf("recv 1: %v", err)

            break

        if err != nil {
            t.Fatalf("recv: %v", err)

    }

    }
}

// Package runtime implements the AgentRuntimeService — the gRPC contract
// kubilitics-backend uses to invoke AI capabilities. v1 (subproject 3a)
// only implements the LLM-direct path. The Router (3b), kagent integration
// (3c), Python multi-agent (3d), and Safety wiring (3e) layer on top
// without changing this contract.
package runtime

import (
    "context"
    "crypto/rand"
    "encoding/hex"
    "fmt"
    "time"

    aiv1 "github.com/kubilitics/kubilitics-ai/api/proto/v1"
    "google.golang.org/grpc/codes"
    "google.golang.org/grpc/status"
    tspb "google.golang.org/protobuf/types/known/timestamppb"
)

// LLMProvider is the minimum interface the v1 runtime needs from the
// existing internal/llm/provider/* implementations. Kept narrow on
// purpose so subprojects 3b-3d can swap in a Router without touching
// this package.
type LLMProvider interface {
    StreamCompletion(ctx context.Context, prompt string) (<-chan string, error)
}

type Config struct {
    LLM           LLMProvider
    AIVersion     string
    SchemaVersion string
    Providers     []string
    Models        []string
}

type Server struct {
    aiv1.UnimplementedAgentRuntimeServiceServer
    cfg Config
}

func New(cfg Config) *Server {
    return &Server{cfg: cfg}
}

func newID(prefix string) string {
    var b [8]byte
    _, _ = rand.Read(b[:])
    return prefix + "-" + hex.EncodeToString(b[:])
}

func (s *Server) Capabilities(_ context.Context, _ *aiv1.Empty) (*aiv1.CapabilitiesResponse, error) {
    return &aiv1.CapabilitiesResponse{
        SchemaVersion: s.cfg.SchemaVersion,
        AiVersion:     s.cfg.AIVersion,
        Engines:       []string{"llm"},
        Providers:     s.cfg.Providers,
        Models:        s.cfg.Models,
        AutonomyLevel: aiv1.AutonomyLevel_AUTONOMY_PROPOSE,
        SupportsUndo:  false,
        SupportsPlans: false,
        AllowedActions: []string{}, // Action gating wired in 3e
    }, nil
}

func (s *Server) Chat(stream aiv1.AgentRuntimeService_ChatServer) error {
    var sessionID string
    for {
        req, err := stream.Recv()
        if err != nil {
            return err
        }
        switch r := req.Request.(type) {
        case *aiv1.ChatRequest_Create:
            sessionID = newID("sess")
            if err := stream.Send(s.event(&aiv1.AIEvent_SessionCreated{
                SessionCreated: &aiv1.SessionCreated{SessionId: sessionID},
            })); err != nil {
                return err
            }

        case *aiv1.ChatRequest_Message:
            if sessionID == "" {
                sessionID = r.Message.SessionId
            }
            if sessionID == "" {
                return status.Error(codes.InvalidArgument, "session_id required (call CreateSession or include in UserMessage)")
            }
            start := time.Now()
            ch, err := s.cfg.LLM.StreamCompletion(stream.Context(), r.Message.Text)
            if err != nil {
                _ = stream.Send(s.event(&aiv1.AIEvent_ErrorEvent{
                    ErrorEvent: &aiv1.ErrorEvent{Code: "llm_error", Message: err.Error()},
                }))
                _ = stream.Send(s.event(&aiv1.AIEvent_Done{
                    Done: &aiv1.Done{
                        EngineUsed:      "llm",
                        TotalLatencyMs:  time.Since(start).Milliseconds(),
                        Cancelled:       false,
                        Partial:         true,
                    },
                }))
                continue
            }
            for chunk := range ch {
                if err := stream.Send(s.event(&aiv1.AIEvent_TextDelta{
                    TextDelta: &aiv1.TextDelta{Text: chunk},
                })); err != nil {
                    return err
                }
            }
            if err := stream.Send(s.event(&aiv1.AIEvent_Done{
                Done: &aiv1.Done{
                    EngineUsed:     "llm",
                    TotalLatencyMs: time.Since(start).Milliseconds(),
                },
            })); err != nil {
                return err
            }

        default:
            return status.Errorf(codes.InvalidArgument, "unknown ChatRequest variant %T", r)
        }
    }
}

func (s *Server) RunAgent(_ aiv1.AgentRuntimeService_RunAgentServer) error {

// event wraps a one-of variant in an AIEvent envelope, stamping

// Package aiclient holds the gRPC + HTTP clients kubilitics-backend uses
// to talk to kubilitics-ai. Replaces the supervisor stack from subproject 2.
package aiclient

import (
    "context"
    "fmt"
    "sync"
    "time"

    aiv1 "github.com/kubilitics/kubilitics-ai/api/proto/v1"
    "google.golang.org/grpc"
    "google.golang.org/grpc/credentials/insecure"
    "google.golang.org/grpc/keepalive"
    "google.golang.org/protobuf/types/known/emptypb"
)

type GRPCClient struct {
    addr   string
    opts   ClientOpts

    mu           sync.Mutex
    conn         *grpc.ClientConn
    client       aiv1.AgentRuntimeServiceClient
    currentState ConnectionState
}

// NewGRPCClient does NOT dial. Call EnsureOpen() before any RPC.
func NewGRPCClient(addr string, opts ClientOpts) *GRPCClient {
    return &GRPCClient{addr: addr, opts: opts, currentState: StateIdle}
}

func (c *GRPCClient) State() ConnectionState {
    c.mu.Lock()
    defer c.mu.Unlock()
    return c.currentState
}

func (c *GRPCClient) ensureOpen(ctx context.Context) error {
    c.mu.Lock()
    if c.conn != nil {
        c.mu.Unlock()
        return nil
    }
    c.currentState = StateConnecting
    c.mu.Unlock()

    dialCtx, cancel := context.WithTimeout(ctx, c.opts.DialTimeout)
    defer cancel()

    conn, err := grpc.NewClient(c.addr,
        grpc.WithTransportCredentials(insecure.NewCredentials()), // in-cluster: TLS terminated by service mesh / mTLS at network layer; gRPC plaintext between same-namespace pods is acceptable for v1
        grpc.WithKeepaliveParams(keepalive.ClientParameters{
            Time:                c.opts.KeepaliveTime,
            Timeout:             c.opts.KeepaliveTimeout,
            PermitWithoutStream: true,
        }),
    )
    if err != nil {
        c.mu.Lock()
        c.currentState = StateError
        c.mu.Unlock()
        return fmt.Errorf("dial %s: %w", c.addr, err)
    }
    // Wait for connection ready or timeout.
    _ = dialCtx
    c.mu.Lock()
    c.conn = conn
    c.client = aiv1.NewAgentRuntimeServiceClient(conn)
    c.currentState = StateOpen
    c.mu.Unlock()
    return nil
}

func (c *GRPCClient) Capabilities(ctx context.Context) (*aiv1.CapabilitiesResponse, error) {
    if err := c.ensureOpen(ctx); err != nil {
        return nil, err
    }
    cctx, cancel := context.WithTimeout(ctx, c.opts.UnaryTimeout)
    defer cancel()
    return c.client.Capabilities(cctx, &emptypb.Empty{})
}

func (c *GRPCClient) Chat(ctx context.Context) (aiv1.AgentRuntimeService_ChatClient, error) {
    if err := c.ensureOpen(ctx); err != nil {
        return nil, err
    }
    cctx, cancel := context.WithTimeout(ctx, c.opts.StreamMaxDuration)
    _ = cancel // stream owns lifetime; cancel fires via deadline
    return c.client.Chat(cctx)
}

func (c *GRPCClient) RunAgent(ctx context.Context) (aiv1.AgentRuntimeService_RunAgentClient, error) {
    if err := c.ensureOpen(ctx); err != nil {
        return nil, err
    }
    cctx, cancel := context.WithTimeout(ctx, c.opts.StreamMaxDuration)
    _ = cancel
    return c.client.RunAgent(cctx)
}

func (c *GRPCClient) CancelTurn(ctx context.Context, sessionOrRunID string, isSession bool) error {
    if err := c.ensureOpen(ctx); err != nil {
        return err
    }
    cctx, cancel := context.WithTimeout(ctx, c.opts.UnaryTimeout)
    defer cancel()
    var req *aiv1.CancelRequest
    if isSession {
        req = &aiv1.CancelRequest{Target: &aiv1.CancelRequest_SessionId{SessionId: sessionOrRunID}}
    } else {
        req = &aiv1.CancelRequest{Target: &aiv1.CancelRequest_RunId{RunId: sessionOrRunID}}
    }
    _, err := c.client.CancelTurn(cctx, req)
    return err
}

func (c *GRPCClient) Close() error {
    c.mu.Lock()
    defer c.mu.Unlock()
    if c.conn == nil {
        return nil
    }
    err := c.conn.Close()
    c.conn = nil
    c.client = nil
    c.currentState = StateIdle
    return err
}

package aiclient

import (
    "context"
    "encoding/json"
    "fmt"
    "net/http"
)

type HTTPClient struct {
    base string
    http *http.Client
    opts ClientOpts
}

type Status struct {
    State   string   `json:"state"`
    Version string   `json:"version"`
    Engines []string `json:"engines"`
}

func NewHTTPClient(baseURL string, opts ClientOpts) *HTTPClient {
    return &HTTPClient{
        base: baseURL,
        http: &http.Client{Timeout: opts.UnaryTimeout},
        opts: opts,
    }
}

func (c *HTTPClient) GetStatus(ctx context.Context) (*Status, error) {
    req, err := http.NewRequestWithContext(ctx, http.MethodGet, c.base+"/status", nil)
    if err != nil {
        return nil, err
    }
    resp, err := c.http.Do(req)
    if err != nil {
        return nil, fmt.Errorf("get status: %w", err)
    }
    defer resp.Body.Close()
    if resp.StatusCode >= 400 {
        return nil, fmt.Errorf("status returned HTTP %d", resp.StatusCode)
    }
    var s Status
    if err := json.NewDecoder(resp.Body).Decode(&s); err != nil {
        return nil, fmt.Errorf("decode status: %w", err)
    }
    return &s, nil
}

// enforces cluster_id presence, injects identity metadata, rate-limits
// per user, and emits prometheus metrics. The underlying transport is
// the aiclient package's gRPC + HTTP clients.

import (
    "context"
    "time"

    "github.com/kubilitics/kubilitics-backend/internal/ai/aiclient"
    "github.com/kubilitics/kubilitics-backend/internal/ai/types"

    aiv1 "github.com/kubilitics/kubilitics-ai/api/proto/v1"
    "google.golang.org/grpc/metadata"
    "google.golang.org/grpc/status"
    "google.golang.org/protobuf/types/known/emptypb"
)

type ctxKey int

const (
    ctxUserID ctxKey = iota
    ctxRequestID
)

func WithUser(ctx context.Context, userID string) context.Context {
    return context.WithValue(ctx, ctxUserID, userID)
}
func WithRequestID(ctx context.Context, reqID string) context.Context {
    return context.WithValue(ctx, ctxRequestID, reqID)
}
func userID(ctx context.Context) string {
    v, _ := ctx.Value(ctxUserID).(string)
    return v
}
func requestID(ctx context.Context) string {
    v, _ := ctx.Value(ctxRequestID).(string)
    return v
}

type Proxy struct {
    grpc *aiclient.GRPCClient
    http *aiclient.HTTPClient
    rl   *rateLimiter
}

func New(g *aiclient.GRPCClient, h *aiclient.HTTPClient, rateLimitPerUserPerMin int) *Proxy {
    return &Proxy{grpc: g, http: h, rl: newRateLimiter(rateLimitPerUserPerMin)}
}

func (p *Proxy) ensureCluster(clusterID string) error {
    if clusterID == "" {
        return types.ErrMissingCluster
    }
    return nil
}

func (p *Proxy) attachMeta(ctx context.Context, clusterID string) context.Context {
    return metadata.AppendToOutgoingContext(ctx,
        "kubilitics-cluster-id", clusterID,
        "kubilitics-user-id", userID(ctx),
        "kubilitics-request-id", requestID(ctx),
    )
}

func (p *Proxy) Capabilities(ctx context.Context, clusterID string) (*aiv1.CapabilitiesResponse, error) {
    if err := p.ensureCluster(clusterID); err != nil {
        proxyErrors.WithLabelValues("capabilities", "missing_cluster").Inc()
        return nil, err
    }
    if uid := userID(ctx); uid != "" && !p.rl.allow(uid) {
        rateDropped.WithLabelValues("capabilities").Inc()
        return nil, types.ErrRateLimited
    }
    start := time.Now()
    resp, err := p.grpc.Capabilities(p.attachMeta(ctx, clusterID))
    proxyDuration.WithLabelValues("capabilities", statusLabel(err)).Observe(time.Since(start).Seconds())
    if err != nil {
        proxyErrors.WithLabelValues("capabilities", statusLabel(err)).Inc()
    }
    return resp, err
}

func (p *Proxy) Chat(ctx context.Context, clusterID string) (aiv1.AgentRuntimeService_ChatClient, error) {
    if err := p.ensureCluster(clusterID); err != nil {
        proxyErrors.WithLabelValues("chat", "missing_cluster").Inc()
        return nil, err
    }
    if uid := userID(ctx); uid != "" && !p.rl.allow(uid) {
        rateDropped.WithLabelValues("chat").Inc()
        return nil, types.ErrRateLimited
    }
    return p.grpc.Chat(p.attachMeta(ctx, clusterID))
}

func (p *Proxy) CancelTurn(ctx context.Context, clusterID, sessionID string) error {
    if err := p.ensureCluster(clusterID); err != nil {
        return err
    }
    return p.grpc.CancelTurn(p.attachMeta(ctx, clusterID), sessionID, true)
}

func (p *Proxy) Status(ctx context.Context) (*aiclient.Status, error) {
    return p.http.GetStatus(ctx)
}

// CreateSession opens a Chat stream just long enough to send CreateSession
// and read back SessionCreated, then closes the send side. The handler
// uses this for `POST /api/v1/ai/sessions`.
func (p *Proxy) CreateSession(ctx context.Context, clusterID, userIDStr, title string) (string, error) {
    if err := p.ensureCluster(clusterID); err != nil {
        return "", err
    }
    stream, err := p.grpc.Chat(p.attachMeta(ctx, clusterID))
    if err != nil {
        return "", err
    }
    defer stream.CloseSend()
    if err := stream.Send(&aiv1.ChatRequest{Request: &aiv1.ChatRequest_Create{
        Create: &aiv1.CreateSession{FocusClusterId: clusterID, UserId: userIDStr, Title: title},
    }}); err != nil {
        return "", err
    }
    ev, err := stream.Recv()
    if err != nil {
        return "", err
    }
    sc := ev.GetSessionCreated()
    if sc == nil {
        return "", types.ErrInternalError("expected SessionCreated, got different event")
    }
    return sc.SessionId, nil
}

func statusLabel(err error) string {
    if err == nil {
        return "ok"
    }
    if s, ok := status.FromError(err); ok {
        return s.Code().String()
    }
    return "error"
}

import (
    aiv1 "github.com/kubilitics/kubilitics-ai/api/proto/v1"

// Expected schema version of AIEvent the backend was built against.
// MUST match the version field in agent_runtime.proto.
const expectedAIEventSchema = "1.0.0"

// aiEventToWSFrame is the SOLE place where engine_used is allowed to be
// read off an AIEvent and is also the SOLE place that strips it before
// forwarding to UI. Other code paths (logging, audit, metrics) read
// engine_used directly off the AIEvent — that's permitted.
//
// Linter rule (added in CI): any reference to ".EngineUsed" outside this
// file fails the build. See .golangci.yml `forbidigo` rule.
func aiEventToWSFrame(ev *aiv1.AIEvent) (wsFrame, error) {
    if ev.SchemaVersion != "" && ev.SchemaVersion != expectedAIEventSchema {
        return wsFrame{
            Type:    "error",
            Payload: jsonString("AIEvent schema mismatch: got " + ev.SchemaVersion + ", expected " + expectedAIEventSchema),
        }, errSchemaMismatch
    }

    switch e := ev.Event.(type) {
    case *aiv1.AIEvent_TextDelta:
        return wsFrame{Type: "text_delta", Payload: jsonAny(map[string]any{
            "anchor_id": ev.AnchorId, "text": e.TextDelta.Text,
        })}, nil

    case *aiv1.AIEvent_Done:
        // STRIP engine_used here. UI never sees it.
        return wsFrame{Type: "done", Payload: jsonAny(map[string]any{
            "anchor_id":         ev.AnchorId,
            "prompt_tokens":     e.Done.PromptTokens,
            "completion_tokens": e.Done.CompletionTokens,
            // NOTE: engine_used omitted on purpose. See file-level comment.
        })}, nil

    case *aiv1.AIEvent_ErrorEvent:
        return wsFrame{Type: "error", Payload: jsonAny(map[string]any{
            "code": e.ErrorEvent.Code, "message": e.ErrorEvent.Message,
        })}, nil

    case *aiv1.AIEvent_SessionCreated:
        // Backend handles SessionCreated internally (CreateSession path);
        // it should not surface as a WS frame to the UI in normal flow.
        // Return a synthetic "ack" or skip. We skip here.
        return wsFrame{Type: "_skip_"}, nil

    // Future-compat: forward unknown variants as their pb.go field name so
    // the UI's UnknownBlock fallback handles them.
    case *aiv1.AIEvent_ToolStart:
        return wsFrame{Type: "tool_start", Payload: jsonProto(e.ToolStart)}, nil
    case *aiv1.AIEvent_ToolEnd:
        return wsFrame{Type: "tool_end", Payload: jsonProto(e.ToolEnd)}, nil
    case *aiv1.AIEvent_ActionPending:
        return wsFrame{Type: "action_pending", Payload: jsonProto(e.ActionPending)}, nil
    case *aiv1.AIEvent_PlanProposed:
        return wsFrame{Type: "plan_proposed", Payload: jsonProto(e.PlanProposed)}, nil
    case *aiv1.AIEvent_Citation:
        return wsFrame{Type: "citation", Payload: jsonProto(e.Citation)}, nil

    default:
        return wsFrame{}, fmt.Errorf("unknown AIEvent variant %T", e)
    }
}

    return &aiv1.CapabilitiesResponse{
        SchemaVersion: "1.0.0",
        AiVersion:     "stub-0.0.1",
        Engines:       []string{"llm"},
        Providers:     []string{"echo"},
        Models:        []string{"echo-1"},
        AutonomyLevel: aiv1.AutonomyLevel_AUTONOMY_PROPOSE,
    }, nil
}

func (s *echoSrv) Chat(stream aiv1.AgentRuntimeService_ChatServer) error {
    for {
        req, err := stream.Recv()
        if err != nil {
            return err
        }
        switch r := req.Request.(type) {
        case *aiv1.ChatRequest_Create:
            stream.Send(&aiv1.AIEvent{
                EmittedAt: tspb.Now(), SchemaVersion: "1.0.0",
                Event: &aiv1.AIEvent_SessionCreated{
                    SessionCreated: &aiv1.SessionCreated{SessionId: "stub-session"},
                },
            })
        case *aiv1.ChatRequest_Message:
            text := r.Message.Text
            // Stream the echo character-by-character to simulate streaming.
            for _, word := range strings.Fields("Echo: " + text) {
                stream.Send(&aiv1.AIEvent{
                    EmittedAt: tspb.Now(), SchemaVersion: "1.0.0",
                    Event: &aiv1.AIEvent_TextDelta{
                        TextDelta: &aiv1.TextDelta{Text: word + " "},
                    },
                })
            }
            stream.Send(&aiv1.AIEvent{
                EmittedAt: tspb.Now(), SchemaVersion: "1.0.0",
                Event: &aiv1.AIEvent_Done{Done: &aiv1.Done{
                    EngineUsed: "stub", PromptTokens: int32(len(text) / 4),
                }},
            })
        }
    }
}

func main() {
    grpcLis, err := net.Listen("tcp", ":50051")
    if err != nil {
        log.Fatalf("grpc listen: %v", err)
    }
    s := grpc.NewServer()
    aiv1.RegisterAgentRuntimeServiceServer(s, &echoSrv{})
    go func() {
        log.Printf("stub kubilitics-ai gRPC on :50051")
        if err := s.Serve(grpcLis); err != nil {
            log.Fatalf("grpc serve: %v", err)
        }
    }()

        json.NewEncoder(w).Encode(map[string]any{
            "state": "ready", "version": "stub-0.0.1", "engines": []string{"llm"},
        })
    })
    log.Printf("stub kubilitics-ai HTTP on :8080")
    log.Fatal(http.ListenAndServe(":8080", nil))
}
